from pathlib import Path html = r''' Emoji Stage

🎭 Emoji Stage

A story that plays itself, one emoji frame at a time.

Story editor

Each block separated by a blank line is one frame. Spaces are part of the stage.

Tip: keep the editor in a monospace font. Add more spaces to move a character.

How it works

The stage is just text. Every frame replaces the previous frame, so changing spaces makes emojis appear to walk.

Example: 

👧          👻

👧       👻

👧    👻

👧  👻
''' path = Path("/mnt/data/index.html") path.write_text(html, encoding="utf-8") print(f"Created: {path}")
